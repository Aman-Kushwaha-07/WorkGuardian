from flask import Blueprint, render_template, request, Response
from flask_login import login_required, current_user
from models import AuditLog, User
from app import db
import csv
from io import StringIO

admin_bp = Blueprint('admin', __name__, url_prefix='/admin')

@admin_bp.route('/dashboard')
@login_required
def admin_dashboard():
    if not current_user.is_admin:
        return "Access Denied", 403
    
    logs = AuditLog.query.order_by(AuditLog.timestamp.desc()).limit(100).all()
    users = User.query.all()
    total_users = User.query.count()
    mfa_enabled_count = User.query.filter_by(mfa_enabled=True).count()
    
    return render_template('admin_dashboard.html', 
                         logs=logs, 
                         users=users, 
                         total_users=total_users,
                         mfa_enabled_count=mfa_enabled_count)

@admin_bp.route('/export-logs')
@login_required
def export_logs():
    if not current_user.is_admin:
        return "Access Denied", 403
    
    logs = AuditLog.query.order_by(AuditLog.timestamp.desc()).all()
    
    output = StringIO()
    writer = csv.writer(output)
    writer.writerow(['Timestamp', 'User ID', 'Event Type', 'Status', 'IP Address'])
    
    for log in logs:
        writer.writerow([log.timestamp, log.user_id, log.event_type, log.status, log.ip_address])
    
    response = Response(output.getvalue(), mimetype='text/csv')
    response.headers.set('Content-Disposition', 'attachment', filename='audit_logs.csv')
    return response

@admin_bp.route('/disable-user/<int:user_id>')
@login_required
def disable_user(user_id):
    if not current_user.is_admin:
        return "Access Denied", 403
    
    user = User.query.get(user_id)
    if user:
        user.is_active = False
        db.session.commit()
    
    return redirect(url_for('admin.admin_dashboard'))
