<script>
function verifyMFA() {
    const otp = document.getElementById('otp').value;
    fetch('/mfa/verify-setup', {
        method: 'POST',
        headers: {'Content-Type': 'application/json'},
        body: JSON.stringify({otp: otp})
    })
    .then(res => res.json())
    .then(data => {
        if(data.success) {
            window.location.href = '/dashboard';
        } else {
            alert('Invalid OTP. Try again.');
        }
    });
}
</script>
